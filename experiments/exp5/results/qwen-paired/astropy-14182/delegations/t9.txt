from .core import DefaultSplitter
from .fixedwidth import (
    FixedWidth,
    FixedWidthData,
    FixedWidthHeader,
    FixedWidthTwoLineDataSplitter,
)


class SimpleRSTHeader(FixedWidthHeader):
    position_line = 0
    start_line = 1
    splitter_class = DefaultSplitter
    position_char = "="

    def get_fixedwidth_params(self, line):
        vals, starts, ends = super().get_fixedwidth_params(line)
        # The right hand column can be unbounded
        ends[-1] = None
        return vals, starts, ends


class SimpleRSTData(FixedWidthData):
    start_line = 3
    end_line = -1
    splitter_class = FixedWidthTwoLineDataSplitter


class RST(FixedWidth):
    """reStructuredText simple format table.

    See: https://docutils.sourceforge.io/docs/ref/rst/restructuredtext.html#simple-tables

    Example::

        ==== ===== ======
        Col1  Col2  Col3
        ==== ===== ======
          1    2.3  Hello
          2    4.5  Worlds
        ==== ===== ======

    Currently there is no support for reading tables which utilize continuation lines,
    or for ones which define column spans through the use of an additional
    line of dashes in the header.

    """

    _format_name = "rst"
    _description = "reStructuredText simple table"
    data_class = SimpleRSTData
    header_class = SimpleRSTHeader

    def __init__(self, header_rows=None):
        super().__init__(delimiter_pad=None, bookend=False, header_rows=header_rows)

    def write(self, lines):
        # If header_rows is None or ['name'], use original behavior for backward compatibility
        if self.header.header_rows is None or self.header.header_rows == ["name"]:
            lines = super().write(lines)
            lines = [lines[1]] + lines + [lines[1]]
            return lines
        
        # Handle custom header_rows case
        header_rows = self.header.header_rows
        data_lines = []
        
        # Get column values for each header row attribute
        hdrs_list = []
        for col_attr in header_rows:
            vals = [
                "" if (val := getattr(col.info, col_attr)) is None else str(val)
                for col in self.data.cols
            ]
            hdrs_list.append(vals)
        
        # Calculate maximum widths for each column
        vals_list = []
        col_str_iters = self.data.str_vals()
        for vals in zip(*col_str_iters):
            vals_list.append(vals)
            
        widths = [
            max(len(vals[i_col]) for vals in vals_list)
            for i_col in range(len(self.data.cols))
        ]
        
        # Incorporate widths for header columns
        if hdrs_list:
            for i_col in range(len(self.data.cols)):
                widths[i_col] = max(
                    widths[i_col], max(len(vals[i_col]) for vals in hdrs_list)
                )
        
        # Create top underline with '=' characters matching column widths
        top_line = [("=" * width) for width in widths]
        lines.append(self.data.splitter.join(top_line, widths))
        
        # Add header rows
        for vals in hdrs_list:
            lines.append(self.data.splitter.join(vals, widths))
        
        # Create bottom underline
        bottom_line = [("=" * width) for width in widths]
        lines.append(self.data.splitter.join(bottom_line, widths))
        
        # Add data rows formatted to match column widths
        for vals in vals_list:
            lines.append(self.data.splitter.join(vals, widths))
            
        return lines
