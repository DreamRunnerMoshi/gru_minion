+def test_case_insensitive_commands():
+    """Test that QDP command parsing is case-insensitive."""
+    from astropy.io.ascii.qdp import _line_type
+    
+    # Test that command parsing is case insensitive
+    commands = [
+        "READ TERR 1",
+        "read terr 1", 
+        "Read Terr 1",
+        "READ serr 2",
+        "read SERR 2",
+        "ReAd SeRr 2"
+    ]
+    
+    # All should be recognized as commands
+    for cmd in commands:
+        result = _line_type(cmd)
+        assert result == "command", f"Failed to recognize command: {cmd}"
+        
+    # Test data lines still work
+    data_line = "1.0 2.0 3.0"
+    result = _line_type(data_line)
+    assert result.startswith("data,"), f"Failed to recognize data line: {data_line}"
+    
+    # Test comment lines still work
+    comment_line = "! This is a comment"
+    result = _line_type(comment_line)
+    assert result == "comment", f"Failed to recognize comment: {comment_line}"
